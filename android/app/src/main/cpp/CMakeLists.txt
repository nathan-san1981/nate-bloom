cmake_minimum_required(VERSION 3.22.1)
project(nbfastpads C CXX)
set(CMAKE_CXX_STANDARD 17)
# Nathan's plugin instruments: the web app's DSP through wasm2c (native/nbsynth); SIMD Everywhere is fetched by CI
set(NBS ${CMAKE_CURRENT_SOURCE_DIR}/../../../../../native/nbsynth)
add_library(nbfastpads SHARED nb_fastpads.cpp ${NBS}/nbsynth.c ${NBS}/wasm-rt-impl.c ${NBS}/wasm-rt-mem-impl.c ${NBS}/wasm-rt-exceptions-impl.c ${NBS}/gen/balladkit.c ${NBS}/gen/duet80.c ${NBS}/gen/fuzzbass.c ${NBS}/gen/hypnagogia.c ${NBS}/gen/nebula80.c ${NBS}/gen/parafour.c ${NBS}/gen/pop89.c ${NBS}/gen/stab85.c ${NBS}/gen/twin77.c ${NBS}/gen/unicorndream.c)
target_include_directories(nbfastpads PRIVATE ${NBS} ${NBS}/gen ${NBS}/simde)
target_compile_definitions(nbfastpads PRIVATE WASM_RT_INSTALL_SIGNAL_HANDLER=0 WASM_RT_USE_MMAP=0 WASM_RT_MEMCHECK_BOUNDS_CHECK=1)
target_compile_options(nbfastpads PRIVATE -O2 -Wall $<$<COMPILE_LANGUAGE:CXX>:-ffast-math> $<$<COMPILE_LANGUAGE:C>:-Wno-unused-function -Wno-unused-variable>)
target_link_libraries(nbfastpads aaudio log m)
# Ableton Link (window.NBAbleton, native/ablink): fetched by CI into native/ablink/link
set(ABL ${CMAKE_CURRENT_SOURCE_DIR}/../../../../../native/ablink)
if(EXISTS ${ABL}/link/AbletonLinkConfig.cmake)
  include(${ABL}/link/AbletonLinkConfig.cmake)
  target_sources(nbfastpads PRIVATE nb_ablink_jni.cpp)
  set_source_files_properties(nb_ablink_jni.cpp PROPERTIES COMPILE_OPTIONS "-fno-fast-math;-fexceptions;-frtti;-Wno-multichar")
  target_include_directories(nbfastpads PRIVATE ${ABL})
  target_link_libraries(nbfastpads Ableton::Link)
else()
  message(WARNING "Ableton Link isn't in native/ablink/link: building without it")
endif()
